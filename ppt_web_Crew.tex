% Slide 2: Brief about the Project
\begin{frame}{Brief about the Project}
    \textbf{Problem:} Students and developers often struggle to find dedicated coding partners with similar skill sets and goals for projects. \\ \vspace{0.5cm}
    \textbf{Solution:} \textit{StudyMate} is a streamlined social platform designed exclusively for developers. \\ \vspace{0.3cm}
    \textbf{Core Features:}
    \begin{itemize}
        \item \textbf{Find Partner:} An algorithm to discover and connect with like-minded developers based on tech stacks.
        \item \textbf{Community Posts:} A dedicated feed to share project updates, ask technical questions, and showcase work.
        \item \textbf{Developer Profiles:} Customizable profiles highlighting individual skills and current focuses.
    \end{itemize}
\end{frame}

% Slide 4: Tools Being Used
\begin{frame}{Tools Being Used}
    This project is being developed using the \textbf{MERN Stack} for a robust, scalable, and responsive application.
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Frontend:} React.js (built with Vite), Tailwind CSS for rapid UI styling.
        \item \textbf{Backend:} Node.js with Express.js for handling API routes and business logic.
        \item \textbf{Database:} MongoDB (NoSQL) for flexible data storage.
        \item \textbf{Authentication:} JSON Web Tokens (JWT) for secure user sessions.
    \end{itemize}
\end{frame}
